<script>
const socket = io();
let cmdCount = 0;

function pad(n){return String(n).padStart(2,'0');}

function updateClock(){
  const now = new Date();
  let h = now.getHours(), m = now.getMinutes(), s = now.getSeconds();
  const ampm = h >= 12 ? 'PM' : 'AM';
  h = h % 12 || 12;
  document.getElementById('clock').textContent = pad(h)+':'+pad(m)+':'+pad(s)+' '+ampm;
  const months = ['JAN','FEB','MAR','APR','MAY','JUN','JUL','AUG','SEP','OCT','NOV','DEC'];
  const now2 = new Date();
  document.getElementById('datestr').textContent = months[now2.getMonth()]+' '+pad(now2.getDate())+' '+now2.getFullYear();
}

function addLog(text, type='active'){
  const log = document.getElementById('cmd-log');
  const line = document.createElement('div');
  line.className = 'log-line '+type;
  line.textContent = text;
  log.appendChild(line);
  log.scrollTop = log.scrollHeight;
}

function addHistory(text){
  const history = document.getElementById('cmd-history');
  const line = document.createElement('div');
  line.className = 'log-line active';
  line.textContent = text;
  history.appendChild(line);
  history.scrollTop = history.scrollHeight;
}

function setMode(mode){
  const statuses = {
    idle: 'STANDING BY',
    listening: 'LISTENING...',
    speaking: 'SPEAKING...',
    processing: 'PROCESSING...',
    alert: 'ALERT!'
  };
  document.getElementById('main-status').textContent = statuses[mode] || 'STANDING BY';
  document.getElementById('orb-status').textContent = mode.toUpperCase();
  addLog('MODE: '+mode.toUpperCase());
}

function sendCommand(){
  const input = document.getElementById('cmd-text');
  const cmd = input.value.trim();
  if(!cmd) return;
  addLog('YOU: '+cmd);
  addHistory(cmd);
  socket.emit('command', {command: cmd});
  input.value = '';
  cmdCount++;
}

function loadStatus(){
  fetch('/api/status')
    .then(r => r.json())
    .then(data => {
      document.getElementById('batt-val').textContent = (data.battery || '--') + ' %';
      document.getElementById('batt-status').textContent = data.plugged ? 'CHARGING' : 'ON BATTERY';
      document.getElementById('weather-val').textContent = data.weather || '--';
      document.getElementById('bot-weather').textContent = data.weather || '--';
      document.getElementById('bot-battery').textContent = (data.battery || '--') + ' %';
      document.getElementById('mem-count').textContent = (data.memory_count || 0) + ' facts';
      document.getElementById('bot-memory').textContent = (data.memory_count || 0) + ' facts';
    })
    .catch(() => {});
}

function loadNews(){
  fetch('/api/news')
    .then(r => r.json())
    .then(data => {
      const container = document.getElementById('news-container');
      container.innerHTML = '';
      data.headlines.forEach((h, i) => {
        const div = document.createElement('div');
        div.className = 'news-item' + (i < 2 ? ' hot' : '');
        div.textContent = h;
        container.appendChild(div);
      });
    })
    .catch(() => {});
}

socket.on('status_update', data => {
  if(data.battery) {
    document.getElementById('batt-val').textContent = data.battery + ' %';
    document.getElementById('bot-battery').textContent = data.battery + ' %';
    document.getElementById('batt-status').textContent = data.plugged ? 'CHARGING' : 'ON BATTERY';
  }
});

socket.on('response', data => {
  addLog('JARVIS: '+data.message, 'jarvis');
});

setInterval(updateClock, 1000);
setInterval(loadStatus, 10000);
updateClock();
loadStatus();
loadNews();
</script>
